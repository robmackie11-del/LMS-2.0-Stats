     function escapeHtml(value) {
       return String(value == null ? '' : value)
         .replace(/&/g, '&')
         .replace(/</g, '<')
         .replace(/>/g, '>')
         .replace(/"/g, '"')
         .replace(/'/g, ''');
     }

     function formatMoney(amount) {
       const num = Number(amount) || 0;
       return '£' + num.toFixed(2);
     }

     function formatProfit(amount) {
       const num = Number(amount) || 0;
       if (num > 0) return '+£' + num.toFixed(2);
       if (num < 0) return '-£' + Math.abs(num).toFixed(2);
       return '£0.00';
     }

     function getProfitClass(amount) {
       const num = Number(amount) || 0;
       if (num > 0) return 'profit-pos';
       if (num < 0) return 'profit-neg';
       return 'profit-zero';
     }

     }

     /* =====================================================================
        4. SAFE URL PARSING & SHARE LINK GENERATION (NEVER THROWS)
        ===================================================================== */
     function parseParamsSafely() {
       try {
         const rawSearch = typeof window.location.search === 'string' ? window.location.search.replace(/^\?/, '') : '';
         const rawHash = typeof window.location.hash === 'string' ? window.location.hash.replace(/^#/, '') : '';
         const combined = new URLSearchParams(rawHash || rawSearch);

         const f = combined.get('filter');
         if (f && VALID_FILTERS.includes(f)) state.filter = f;

         const q = combined.get('q');
         if (q !== null) state.query = q.slice(0, 80);

         const s = combined.get('sort');
         if (s && VALID_SORTS.includes(s)) state.sortBy = s;

         const d = combined.get('dir');
         if (d === 'asc' || d === 'desc') state.sortDir = d;

         const shot = combined.get('shot');
         if (shot === '1' || shot === 'true') state.screenshotMode = true;
       } catch (_err) {
         // Ignore malformed or restricted URLs in sandboxed/content views
       }
     }

     function buildShareHash() {
       try {
         const params = new URLSearchParams();
         if (state.filter !== 'all') params.set('filter', state.filter);
         if (state.query.trim()) params.set('q', state.query.trim());
         if (state.sortBy !== 'rank') params.set('sort', state.sortBy);
         if (state.sortDir !== 'desc') params.set('dir', state.sortDir);
         if (state.screenshotMode) params.set('shot', '1');
         const str = params.toString();
         return str ? '#' + str : '';
       } catch (_err) {
         return '';
       }
     }

     function syncHashSafely() {
       try {
         const nextHash = buildShareHash();
         if (window.location.hash !== nextHash) {
           window.location.hash = nextHash ? nextHash.slice(1) : '';
         }
       } catch (_err) {
         // Safe fallback when hash assignment is restricted
       }
     }

     function getShareUrl() {
       const hash = buildShareHash();
       try {
         const href = String(window.location.href || '');
         if (href && !href.startsWith('about:blank') && !href.startsWith('about:srcdoc')) {
           const noHash = href.split('#')[0];
           return noHash + hash;
         }
       } catch (_err) {
         // Fallback below
       }
       return hash || '#all';
     }

       const url = getShareUrl();

       function finishSuccess() {

           els.copyLinkLabel.textContent = 'Link Copied!';

       function fallbackCopy() {
         try {
           const ta = document.createElement('textarea');
           ta.value = url;
           ta.setAttribute('readonly', '');
           ta.style.position = 'fixed';
           ta.style.opacity = '0';
           document.body.appendChild(ta);
           ta.focus();
           ta.select();
           const ok = document.execCommand('copy');
           document.body.removeChild(ta);
           if (ok) {
             finishSuccess();
           } else {
             showBanner('Copy link manually: ' + url, true);
           }
         } catch (_err) {
           showBanner('Copy link manually: ' + url, true);
         }
       }

       try {
         if (navigator.clipboard && typeof navigator.clipboard.writeText === 'function') {
           navigator.clipboard.writeText(url).then(finishSuccess).catch(fallbackCopy);
         } else {
           fallbackCopy();
         }
       } catch (_err) {
         fallbackCopy();
       }
     }

     function getFilteredSortedPlayers() {
       const q = state.query.trim().toLowerCase();
       const dirFactor = state.sortDir === 'asc' ? -1 : 1;

       const filtered = PLAYERS.filter(function (p) {
         if (state.filter === 'winners' && p.wins <= 0) return false;
         if (state.filter === 'nonwinners' && p.wins > 0) return false;

         if (q) {
           const haystack = [
             p.name,
             p.bestTeam,
             p.bestRound,
             p.bestOdds || '',
             p.bestNote || ''
           ].join(' ').toLowerCase();
           if (!haystack.includes(q)) return false;
         }
         return true;
       });

       filtered.sort(function (a, b) {
         if (state.sortBy === 'rank') {
           // Default order keeps official rank 1..10 when DESC, or 10..1 when ASC
           return (a.rank - b.rank) * dirFactor;
         }

         let diff = 0;
         switch (state.sortBy) {
           case 'winnings':
             diff = (b.winnings - a.winnings) || (b.netProfit - a.netProfit);
             break;
           case 'profit':
             diff = (b.netProfit - a.netProfit) || (b.winnings - a.winnings);
             break;
           case 'bestRound':
             diff = b.bestPoints - a.bestPoints;
             break;
           case 'correctPicks':
             diff = (b.correctPicks - a.correctPicks) || (a.losingPicks - b.losingPicks);
             break;
           case 'losingPicks':
             diff = (b.losingPicks - a.losingPicks) || (b.correctPicks - a.correctPicks);
             break;
           case 'apps':
             diff = (b.apps - a.apps) || (b.netProfit - a.netProfit);
             break;
           case 'wins':
             diff = (b.wins - a.wins) || (b.winnings - a.winnings);
             break;
           default:
             diff = a.rank - b.rank;
         }

         return (diff * dirFactor) || (a.rank - b.rank);
       });

       return filtered;
     }

     function renderTable() {
       if (!els.standingsBody) return;
       const rows = getFilteredSortedPlayers();

       if (!rows.length) {
         els.standingsBody.innerHTML =
           '<tr><td colspan="8" class="empty-state">No players match your current filter or search. Click <strong>Reset</strong> to view all 10 players.</td></tr>';
         return;
       }

       const html = rows.map(function (p) {
         const badgeHtml = p.wins > 0
           ? '<span class="win-badge"><span aria-hidden="true">&#127942;</span><span>' + escapeHtml(p.wins) + (p.wins === 1 ? ' Win' : ' Wins') + '</span></span>'
           : '';

         return (
           '<tr>' +
             '<td>' +
               '<div class="player-cell">' +
                 '<span class="player-name">' + escapeHtml(p.name) + '</span>' +
                 badgeHtml +
               '</div>' +
             '</td>' +
             '<td class="num">' + escapeHtml(p.apps) + '</td>' +
             '<td class="num">' + escapeHtml(p.wins) + '</td>' +
             '<td class="num winnings-val">' + escapeHtml(formatMoney(p.winnings)) + '</td>' +
             '<td class="num ' + getProfitClass(p.netProfit) + '">' + escapeHtml(formatProfit(p.netProfit)) + '</td>' +
             '<td class="num picks-correct">' + escapeHtml(p.correctPicks) + '</td>' +
             '<td class="num picks-losing">' + escapeHtml(p.losingPicks) + '</td>' +
             '<td>' +
               '<div class="best-round-box">' +
                 '<div class="best-round-main">' + escapeHtml(p.bestTeam) + ' (' + escapeHtml(Number(p.bestPoints).toFixed(2)) + ' pts)</div>' +
                 '<div class="best-round-meta">' + formatBestMeta(p) + '</div>' +
               '</div>' +
             '</td>' +
           '</tr>'
         );
       }).join('');

       els.standingsBody.innerHTML = html;
     }

        ===================================================================== */
     function bindEvents() {
       if (els.searchInput) {
         els.searchInput.addEventListener('input', function (e) {
           state.query = e.target.value || '';
           updateView(true);
         });
       }

       els.filterPills.forEach(function (pill) {
         pill.addEventListener('click', function () {
           const nextFilter = pill.getAttribute('data-filter');
           if (nextFilter && VALID_FILTERS.includes(nextFilter)) {
             state.filter = nextFilter;
             updateView(true);
           }
         });
       });

       if (els.sortSelect) {
         els.sortSelect.addEventListener('change', function (e) {
           const nextSort = e.target.value;
           if (VALID_SORTS.includes(nextSort)) {
             state.sortBy = nextSort;
             updateView(true);
           }
         });
       }

       if (els.sortDirBtn) {
         els.sortDirBtn.addEventListener('click', function () {
           state.sortDir = state.sortDir === 'desc' ? 'asc' : 'desc';
           updateView(true);
         });
       }

       if (els.resetBtn) {
         els.resetBtn.addEventListener('click', function () {
           state.filter = 'all';
           state.query = '';
           state.sortBy = 'rank';
           state.sortDir = 'desc';
           state.screenshotMode = false;
           hideBanner();
           updateView(true);
         });
       }

       if (els.copyLinkBtn) {
         els.copyLinkBtn.addEventListener('click', copyShareLink);
       }

       if (els.shotModeBtn) {
         els.shotModeBtn.addEventListener('click', function () {
           state.screenshotMode = !state.screenshotMode;
           updateView(true);
         });
       }

       if (els.exitShotFloatBtn) {
         els.exitShotFloatBtn.addEventListener('click', function () {
           state.screenshotMode = false;
           updateView(true);
         });
       }

       if (els.dismissBannerBtn) {
         els.dismissBannerBtn.addEventListener('click', hideBanner);
       }

       window.addEventListener('hashchange', function () {
         state.filter = 'all';
         state.query = '';
         state.sortBy = 'rank';
         state.sortDir = 'desc';
         state.screenshotMode = false;
         parseParamsSafely();
         updateView(false);
       });
     }

     function init() {
       try {
         parseParamsSafely();
         bindEvents();
         updateView(false);
       } catch (err) {
         if (els.standingsBody) {
           els.standingsBody.innerHTML =
             '<tr><td colspan="8" class="empty-state">Unable to initialize standings view: ' + escapeHtml(err && err.message ? err.message : 'Unknown error') + '</td></tr>';
         }
       }
     }

     if (document.readyState === 'loading') {
       document.addEventListener('DOMContentLoaded', init);
     } else {
       init();
     }
   })();
 </script>
